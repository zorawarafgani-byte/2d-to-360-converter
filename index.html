async function generate3D() { const apiKey = document.getElementById('apiKeyInput').value.trim(); const statusText = document.getElementById('statusText'); if (!apiKey) { alert('Please enter your Tripo3D API Key!'); return; } if (!selectedFile) { alert('Please select an image first!'); return; } statusText.innerText = "Processing image and requesting AI 3D model..."; const reader = new FileReader(); reader.readAsDataURL(selectedFile); reader.onload = async function () { const imageBase64 = reader.result; try { const response = await fetch('/api/convert', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ apiKey, imageBase64 }) }); const data = await response.json(); if (!response.ok) throw new Error(data.message || 'Generation failed'); document.getElementById('modelViewer').src = data.modelUrl; document.getElementById('viewerContainer').style.display = 'block'; statusText.innerText = "✨ 3D Model Generated Successfully!"; } catch (err) { statusText.innerText = "Error: " + err.message; } }; }
